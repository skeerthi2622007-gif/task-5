# Weather Prediction using Bayesian Probability

# Dataset
total_days = 20

rain_days = 8
not_rain_days = 12

cloudy_rain_days = 7
cloudy_not_rain_days = 3

# Prior probabilities
p_rain = rain_days / total_days
p_not_rain = not_rain_days / total_days

# Conditional probabilities
p_cloudy_given_rain = cloudy_rain_days / rain_days
p_cloudy_given_not_rain = cloudy_not_rain_days / not_rain_days

# Probability of Cloudy
p_cloudy = (
    p_cloudy_given_rain * p_rain
    + p_cloudy_given_not_rain * p_not_rain
)

# Bayes' Theorem
p_rain_given_cloudy = (
    p_cloudy_given_rain * p_rain
) / p_cloudy

p_not_rain_given_cloudy = (
    p_cloudy_given_not_rain * p_not_rain
) / p_cloudy

# Display results
print("WEATHER PREDICTION USING BAYESIAN PROBABILITY")
print("----------------------------------------------")

print("P(Rain) =", round(p_rain, 3))
print("P(Not Rain) =", round(p_not_rain, 3))

print("P(Cloudy | Rain) =", round(p_cloudy_given_rain, 3))
print("P(Cloudy | Not Rain) =", round(p_cloudy_given_not_rain, 3))

print("P(Rain | Cloudy) =", round(p_rain_given_cloudy, 3))
print("P(Not Rain | Cloudy) =", round(p_not_rain_given_cloudy, 3))

# Prediction
if p_rain_given_cloudy > p_not_rain_given_cloudy:
    prediction = "Rain"
else:
    prediction = "Not Rain"

print("Predicted Weather =", prediction)
